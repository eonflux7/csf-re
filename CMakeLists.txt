cmake_minimum_required(VERSION 3.20)
project(csf_re C CXX)
# Build: cmake -B build -G Ninja -DCMAKE_TOOLCHAIN_FILE=cmake/clang-cl-x86.cmake && ninja -C build
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")  # static CRT: no runtime DLLs needed on the game box

file(GLOB_RECURSE CSF_SRC CONFIGURE_DEPENDS src/*.cpp)
# Cutover-only runtime/library boundaries; linktest.py enumerates these separately.
list(FILTER CSF_SRC EXCLUDE REGEX "/boundary/")
if(NOT EXISTS ${CMAKE_BINARY_DIR}/diff_table.inc)  # regenerated by tools/deploy.sh (tools/difftable.py)
  file(WRITE ${CMAKE_BINARY_DIR}/diff_table.inc "{0}\n")
endif()
if(NOT EXISTS ${CMAKE_BINARY_DIR}/replay_table.inc)  # tier 3, also written by tools/difftable.py
  file(WRITE ${CMAKE_BINARY_DIR}/replay_table.inc "\n")
endif()
# MinHook (BSD-2, vendored): trampolines for wrap-and-call-original hooks and hit-count probes
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
            third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde32.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
add_library(csf_hook SHARED hook/main.cpp hook/autopilot.cpp hook/diff.cpp hook/replay.cpp hook/dinput8.def ${CSF_SRC})
set_target_properties(csf_hook PROPERTIES OUTPUT_NAME dinput8)
target_include_directories(csf_hook PRIVATE hook src ${CMAKE_BINARY_DIR})  # build/: diff_table.inc
target_compile_options(csf_hook PRIVATE /W3 /EHsc /Zc:threadSafeInit- -Wno-microsoft-cast -Wno-invalid-offsetof)
target_link_libraries(csf_hook PRIVATE minhook kernel32 user32 ole32)
target_link_options(csf_hook PRIVATE /MAP:${CMAKE_BINARY_DIR}/dinput8.map)

# The cutover target needs the librw checkout (work/tools/librw). Trees without it (the verifier's) build the
# hooked DLL only.
if(EXISTS "${CMAKE_SOURCE_DIR}/work/tools/librw/CMakeLists.txt")
  include(cmake/standalone.cmake)
endif()
# Opt-in cutover boundary. The hook DLL never links librw or boundary slots.
option(CSF_RW_CPU_SLICE "Build the standalone librw CPU boundary and link probe" OFF)
if(CSF_RW_CPU_SLICE)
  add_subdirectory(src/boundary/rw)
endif()
