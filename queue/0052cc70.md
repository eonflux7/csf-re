// Rounds (hi * 2^64 + lo) * 2^exponent to a 64-bit significand, round-half-even.
static MenuProgressInput_Ext MenuProgressInput_ExtRound(bool negative, uint64_t hi, uint64_t lo, int32_t exponent)
{

        lo = 0;
        exponent += 64;

        lo <<= 1;
        --exponent;

    }

// subnormals at 2^-149, overflow to infinity, NaN payload truncated).
static float MenuProgressInput_ExtToFloat(MenuProgressInput_Ext value)
{

    const uint32_t exponentAllOnes = static_cast<uint32_t>(0xFFu) << 23;

    if (value.kind == kExtInfinity)
    {
        return MenuProgressInput_BitsToFloat(signBit | exponentAllOnes);
    }
    if (value.kind == kExtNaN)
    {
        // Significand bits 62..40 become the 23 fraction bits.
        const uint32_t fraction = static_cast<uint32_t>((value.mantissa << 1) >> 41);
        return MenuProgressInput_BitsToFloat(signBit | exponentAllOnes | fraction);
    }
    if (value.mantissa == 0)
    {
        return MenuProgressInput_BitsToFloat(signBit);
    }

    // The leading bit has weight 2^(exponent + 63). A float keeps 24 bits, so its LSB
    // has weight 2^(exponent + 40), but never below 2^-149.
    int32_t lsbExponent = value.exponent + 40;
    if (lsbExponent < -149)
    {
        lsbExponent = -149;
    }
    const int32_t shift = lsbExponent - value.exponent;
    if (shift > 64)
    {
        return MenuProgressInput_BitsToFloat(signBit);
    }

    uint64_t kept;
    if (shift == 64)
    {
        kept = (value.mantissa > (uint64_t(1) << 63)) ? 1 : 0;
    }
    else
    {
        kept = value.mantissa >> shift;
        const uint64_t dropped = value.mantissa & ((uint64_t(1) << shift) - 1);
        const uint64_t half = uint64_t(1) << (shift - 1);
        if (dropped > half || (dropped == half && (kept & 1) != 0))
        {
            ++kept;
        }
    }

    if (kept == 0)
    {
        return MenuProgressInput_BitsToFloat(signBit);
    }
    if ((kept >> 24) != 0)
    {
        // Rounding carried out of the 24-bit significand.
        kept >>= 1;
        ++lsbExponent;
    }

    uint32_t bits;
    if ((kept >> 23) == 0)
    {
        // Subnormal: lsbExponent is -149 here, so the significand is the encoding.
        bits = static_cast<uint32_t>(kept);
    }
    else
    {
        const int32_t biasedExponent = lsbExponent + 150;
        if (biasedExponent >= 255)
        {
            bits = exponentAllOnes; // overflow to infinity
        }
        else
        {
            // kept has its implicit bit (bit 23) set; adding it to (biased - 1) << 23
            // yields (biased << 23) | fraction.
            bits = (static_cast<uint32_t>(biasedExponent - 1) << 23) + static_cast<uint32_t>(kept);
        }
    }
    return MenuProgressInput_BitsToFloat(bits | signBit);
}
