# Use the checked-out MIT project, not copied sources or RenderWare SDK headers.
include("${CMAKE_SOURCE_DIR}/cmake/librw.cmake")
add_library(csf_rw_cpu STATIC cpu.cpp life.cpp)
target_compile_definitions(csf_rw_cpu PRIVATE CSF_STANDALONE)
target_include_directories(csf_rw_cpu PRIVATE "${CMAKE_SOURCE_DIR}/src" "${CMAKE_SOURCE_DIR}/hook")
if(WIN32)
  target_compile_options(csf_rw_cpu PRIVATE /W3 /EHsc /Zc:threadSafeInit- -Wno-microsoft-cast)
else()
  target_compile_definitions(csf_rw_cpu PRIVATE CSF_RW_NATIVE_TEST)
  target_compile_options(csf_rw_cpu PRIVATE -fms-extensions -Wno-ignored-attributes)
endif()
target_link_libraries(csf_rw_cpu PUBLIC librw::librw)
add_executable(csf_rw_cpu_link "${CMAKE_SOURCE_DIR}/tests/rw_cpu.cpp" "${CMAKE_SOURCE_DIR}/tests/rw_globals.cpp")
target_compile_definitions(csf_rw_cpu_link PRIVATE CSF_STANDALONE)
target_include_directories(csf_rw_cpu_link PRIVATE "${CMAKE_SOURCE_DIR}/src" "${CMAKE_SOURCE_DIR}/hook")
target_link_libraries(csf_rw_cpu_link PRIVATE csf_rw_cpu)
if(WIN32)
  target_compile_options(csf_rw_cpu_link PRIVATE /UNDEBUG)
else()
  target_compile_options(csf_rw_cpu_link PRIVATE -UNDEBUG)
  target_compile_options(csf_rw_cpu_link PRIVATE -fms-extensions -Wno-ignored-attributes)
endif()
add_executable(csf_rw_life_link "${CMAKE_SOURCE_DIR}/tests/rw_life.cpp" "${CMAKE_SOURCE_DIR}/tests/rw_globals.cpp")
target_compile_definitions(csf_rw_life_link PRIVATE CSF_STANDALONE)
target_include_directories(csf_rw_life_link PRIVATE "${CMAKE_SOURCE_DIR}/src" "${CMAKE_SOURCE_DIR}/hook")
target_link_libraries(csf_rw_life_link PRIVATE csf_rw_cpu)
if(WIN32)
  target_compile_options(csf_rw_life_link PRIVATE /UNDEBUG)
else()
  target_compile_options(csf_rw_life_link PRIVATE -UNDEBUG -fms-extensions -Wno-ignored-attributes)
endif()
